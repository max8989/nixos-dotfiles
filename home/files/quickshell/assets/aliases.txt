# NixOS Cheatsheet
# Open with Super + /

## Windows, panes and editor splits
Super + H/J/K/L           # navigate Hyprland windows
Alt + H/J/K/L             # navigate Herdr panes
Ctrl + H/J/K/L            # navigate Neovim splits (and tmux panes inside tmux)

## Herdr Desktop
Ctrl + 1..9               # select Herdr tab
Ctrl + Shift + 1..9       # select Herdr space; avoids Hyprland's Alt-number screenshot keys
Ctrl + Shift + T          # create Herdr tab
Ctrl + Shift + N          # create Herdr space
Ctrl + Shift + Enter      # open a new terminal pane to the right (same as Ctrl + Shift + O)
Ctrl + Shift + O          # split Herdr pane right
Ctrl + Shift + E          # split Herdr pane down
Ctrl + Shift + Z          # zoom Herdr pane
Ctrl + Shift + R          # rename Herdr tab; F2 passes through to terminal apps
Alt + Shift + H/J/K/L     # move Herdr pane left/down/up/right

## LazyVim in Herdr
[e / ]e                   # move line or linewise visual selection up/down
Ctrl + X, Ctrl + O        # trigger completion in Insert mode; Ctrl-Space switches input language
F2 / Space c r / Space r n # rename symbol with an attached language server
Shift + H/L               # previous/next Neovim buffer
Space b b                 # switch to the other Neovim buffer

## Find files and folders (fd)
ff <name>                 # find files and folders by name, recursive
fd1 <name>                # same, current directory only
fdd <name>                # folders only
fd -t f <name>            # files only
fd -e <ext>               # by extension, ex: fd -e nix
fz                        # fd piped into fzf, pick a file interactively

## Find text inside files (ripgrep)
ft <text>                 # search text in files, recursive
ft1 <text>                # same, current directory only
ft -l <text>              # only list the file names
ft -i <text>              # ignore case
ft -g '*.nix' <text>      # only search files matching a glob
fif [text]                # live search: type, preview, Enter opens nvim at the line

## fzf keys
Ctrl + T                  # pick a file and paste its path
Ctrl + R                  # search command history
Alt + C                   # pick a folder and cd into it
**<Tab>                   # fuzzy completion, ex: nvim **<Tab>

## Listing
ls                        # eza with icons, folders first
ll                        # long list
la                        # long list with hidden files
lt                        # tree view
cat <file>                # bat, no pager

## Navigation
..                        # cd ..
...                       # cd ../..
z <part of path>          # jump to a frequent folder (zoxide)
zi                        # pick a frequent folder with fzf

## Editing and notes
n                         # nvim in the current folder
note                      # open the Obsidian notes in nvim
an                        # audio-notes

## Document conversion (Docling)
docling document.pdf --to md                              # convert to Markdown
docling document.pdf --to text                            # convert to plain text
docling document.pdf --to html                            # convert to HTML
docling document.pdf --to json                            # convert to structured JSON
docling document.pdf --to md --to json                    # export Markdown and JSON together
docling document.pdf --to md --output ./converted         # save Markdown in a chosen folder
docling ./documents --to md --output ./converted          # convert a folder of documents to Markdown

## Clipboard helpers
pwdc                      # copy the current path
cdc                       # copy "cd <here> && claude"
cdo                       # copy "cd <here> && opencode"
cdd                       # copy "cd <here>"

## Project NOMAD (portable drive)
xdg-open http://localhost:8080                       # open NOMAD on the laptop (server: http://192.168.2.43:8080)
systemctl status project-nomad                       # is NOMAD running?
sudo systemctl stop project-nomad                    # stop NOMAD and its apps, drive stays mounted
sudo systemctl start project-nomad                   # start NOMAD and its apps again
sudo systemctl stop 'opt-project\x2dnomad.mount'     # BEFORE UNPLUGGING: stop NOMAD, then unmount the drive
findmnt /opt/project-nomad                           # prints nothing = safe to unplug
sudo docker compose -p project-nomad -f /opt/project-nomad/compose.yml pull && sudo systemctl restart project-nomad # update NOMAD (needs internet); apps update from its UI
journalctl -u project-nomad -e                       # NOMAD start/stop logs

## NixOS updates (homeserver prebuilds nightly)
nixos-update                                    # pull the homeserver's last good flake.lock and switch; heavy packages download
bump-latest                                     # nix flake update right now and switch; new packages compile locally
nixos-switch                                    # switch to the checked-out config; uses the cache when it has the build
ssh maxime@192.168.2.43 systemctl status prebuild-desktop     # last nightly prebuild: running, done or failed?
ssh maxime@192.168.2.43 journalctl -u prebuild-desktop -n 30  # last nightly prebuild log
ssh -t maxime@192.168.2.43 sudo systemctl start --no-block prebuild-desktop # run the prebuild now, in the background

## Misc
c                       # clear
q                         # exit
df / du / free            # human readable sizes
neofetch                  # fastfetch
